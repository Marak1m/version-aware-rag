from datetime import datetime

from pydantic import BaseModel, field_serializer


class Round(BaseModel):
    played: datetime
    points: int

    @field_serializer("played")
    def day(self, value):
        return value.date().isoformat()


print(Round(played=datetime(2024, 3, 1), points=3))
