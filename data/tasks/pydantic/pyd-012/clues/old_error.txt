from pydantic import BaseModel, model_validator


class Login(BaseModel):
    secret: str
    confirm: str

    @model_validator(mode="after")
    def same(self):
        if self.secret != self.confirm:
            raise ValueError("secrets differ")
        return self


print(Login(secret="hunter22", confirm="hunter22"))
