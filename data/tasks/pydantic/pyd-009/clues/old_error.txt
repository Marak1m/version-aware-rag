from pydantic import BaseModel, computed_field


class Contact(BaseModel):
    first: str
    last: str

    @computed_field
    @property
    def full(self) -> str:
        return f"{self.first} {self.last}"


print(Contact(first="Ada", last="Lovelace").full)
