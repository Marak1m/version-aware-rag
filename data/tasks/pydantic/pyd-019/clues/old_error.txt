from datetime import datetime

from pydantic import BaseModel, field_serializer


class Stamp(BaseModel):
    at: datetime

    @field_serializer("at")
    def iso(self, value):
        return value.isoformat()


print(Stamp(at=datetime(2024, 1, 1)))
