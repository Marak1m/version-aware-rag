from pydantic import BaseModel, field_validator


class Product(BaseModel):
    name: str
    price: float

    @field_validator("price")
    @classmethod
    def positive(cls, value):
        if value <= 0:
            raise ValueError("price must be positive")
        return value


print(Product(name="pen", price=1.5))
